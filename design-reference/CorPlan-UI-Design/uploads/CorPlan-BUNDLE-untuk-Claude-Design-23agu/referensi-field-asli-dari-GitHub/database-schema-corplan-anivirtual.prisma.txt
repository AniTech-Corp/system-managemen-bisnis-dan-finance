model CorPlanProject {
  id          String   @id @default(uuid()) @db.Uuid
  workspaceId String   @map("workspace_id") @db.Uuid
  projectId   String?  @map("project_id") @db.Uuid
  title       String   @db.VarChar(255)
  revisionSeq Int      @default(1) @map("revision_seq")
  createdAt   DateTime @default(now()) @map("created_at")
  updatedAt   DateTime @updatedAt @map("updated_at")

  workspace   Workspace       @relation(fields: [workspaceId], references: [id], onDelete: Cascade)
  project     Project?        @relation(fields: [projectId], references: [id], onDelete: SetNull)
  outputs     CorPlanOutput[]

  @@index([workspaceId])
  @@map("corplan_projects")
}

  outputData      Json           @map("output_data") @db.JsonB
  createdAt       DateTime       @default(now()) @map("created_at")

  corplanProject  CorPlanProject @relation(fields: [corplanProjectId], references: [id], onDelete: Cascade)

  @@index([corplanProjectId, module])
  @@map("corplan_outputs")
}
